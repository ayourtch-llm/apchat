[test]
use apchat_models::Message;

fn test_validate_history() {
    // Test with empty history
    let messages: Vec<Message> = vec![];
    assert!(src::chat::history::validate_history(&messages).is_ok());

    // Test with valid history
    let messages = vec![
        Message {
            role: "system".to_string(),
            content: "You are a helpful assistant".to_string(),
            tool_calls: None,
            tool_call_id: None,
            name: None,
            reasoning: None,
        },
        Message {
            role: "user".to_string(),
            content: "Hello".to_string(),
            tool_calls: None,
            tool_call_id: None,
            name: None,
            reasoning: None,
        },
        Message {
            role: "assistant".to_string(),
            content: "Hi there!".to_string(),
            tool_calls: None,
            tool_call_id: None,
            name: None,
            reasoning: None,
        },
    ];
    assert!(src::chat::history::validate_history(&messages).is_ok());

    // Test with invalid role
    let messages = vec![
        Message {
            role: "invalid".to_string(),
            content: "Test".to_string(),
            tool_calls: None,
            tool_call_id: None,
            name: None,
            reasoning: None,
        },
    ];
    assert!(src::chat::history::validate_history(&messages).is_err());
}

fn test_fix_interrupted_history() {
    // Test assistant -> assistant fix
    let messages = vec![
        Message {
            role: "user".to_string(),
            content: "Hello".to_string(),
            tool_calls: None,
            tool_call_id: None,
            name: None,
            reasoning: None,
        },
        Message {
            role: "assistant".to_string(),
            content: "Hi".to_string(),
            tool_calls: None,
            tool_call_id: None,
            name: None,
            reasoning: None,
        },
        Message {
            role: "assistant".to_string(),
            content: "How are you?".to_string(),
            tool_calls: None,
            tool_call_id: None,
            name: None,
            reasoning: None,
        },
    ];
    let (fixed, changed) = src::chat::history::fix_interrupted_history(&messages);
    assert!(changed);
    assert_eq!(fixed.len(), 4);
    assert_eq!(fixed[2].role, "user");
}

fn test_insert_bogus_message() {
    let messages = vec![
        Message {
            role: "user".to_string(),
            content: "Hello".to_string(),
            tool_calls: None,
            tool_call_id: None,
            name: None,
            reasoning: None,
        },
    ];
    let fixed = src::chat::history::insert_bogus_message(&messages, 1, "assistant", "recovery point");
    assert_eq!(fixed.len(), 2);
    assert_eq!(fixed[1].role, "assistant");
    assert!(fixed[1].content.contains("[BOGUS:"));
}
